-- Services
local Players = game:GetService("Players")
local CoreGui = game:GetService("CoreGui")
local UserInputService = game:GetService("UserInputService")
local StarterGui = game:GetService("StarterGui")
local VirtualUser = game:GetService("VirtualUser")
local TeleportService = game:GetService("TeleportService")

local player = Players.LocalPlayer
local playerGui = player:WaitForChild("PlayerGui")

-- Safety notification function
local function sendNotification(title, text)
	pcall(function()
		StarterGui:SetCore("SendNotification", {
			Title = title,
			Text = text,
			Duration = 2.5
		})
	end)
end

-- Create ScreenGui
local screenGui = Instance.new("ScreenGui")
screenGui.Name = "AntiAFKGui"
screenGui.ResetOnSpawn = false
screenGui.ZIndexBehavior = Enum.ZIndexBehavior.Sibling

pcall(function()
	screenGui.Parent = CoreGui
end)
if not screenGui.Parent then
	screenGui.Parent = playerGui
end

-- Create Circular Logo (ImageButton)
local logoButton = Instance.new("ImageButton")
logoButton.Name = "AntiAFKLogo"
logoButton.Size = UDim2.new(0, 60, 0, 60)
logoButton.Position = UDim2.new(0.85, 0, 0.2, 0)
logoButton.BackgroundColor3 = Color3.fromRGB(40, 40, 40)
logoButton.Image = "rbxassetid://6031097225" -- Palitan ng iyong rbxassetid photo
logoButton.ScaleType = Enum.ScaleType.Fit
logoButton.AutoButtonColor = true
logoButton.Parent = screenGui

-- UICorner para maging pabilog
local uiCorner = Instance.new("UICorner")
uiCorner.CornerRadius = UDim.new(1, 0)
uiCorner.Parent = logoButton

-- UIStroke para sa kulay (Red = Off, Green = On)
local uiStroke = Instance.new("UIStroke")
uiStroke.Thickness = 3
uiStroke.Color = Color3.fromRGB(200, 50, 50)
uiStroke.Parent = logoButton

-- Anti-AFK & Auto-Reconn Logic
local antiAfkActive = false
local antiAfkConnection = nil
local errorGuiConnection = nil

local function toggleAntiAfk()
	antiAfkActive = not antiAfkActive

	if antiAfkActive then
		uiStroke.Color = Color3.fromRGB(50, 200, 50)
		sendNotification("Anti-AFK", "Anti-AFK is now TURNED ON!")

		-- Ligtas na Anti-AFK simulation para hindi ma-flag as cheat
		local idledConnection = player.Idled:Connect(function()
			VirtualUser:CaptureController()
			VirtualUser:ClickButton1(Vector2.new(0,0))
		end)

		antiAfkConnection = idledConnection
		
		-- Auto-Rejoin kapag na-kick o nadiskonekta
		errorGuiConnection = GuiService.ErrorMessageChanged:Connect(function()
			task.wait(1)
			TeleportService:Teleport(game.PlaceId, player)
		end)
	else
		uiStroke.Color = Color3.fromRGB(200, 50, 50)
		sendNotification("Anti-AFK", "Anti-AFK is now TURNED OFF!")

		if antiAfkConnection then
			antiAfkConnection:Disconnect()
			antiAfkConnection = nil
		end
		if errorGuiConnection then
			errorGuiConnection:Disconnect()
			errorGuiConnection = nil
		end
	end
end

-- Click Event para i-on/off
logoButton.MouseButton1Click:Connect(toggleAntiAfk)

-- Draggable Functionality (Mouse & Touch)
local dragging = false
local dragInput, dragStart, startPos

logoButton.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		dragging = true
		dragStart = input.Position
		startPos = logoButton.Position

		input.Changed:Connect(function()
			if input.UserInputState == Enum.UserInputState.End then
				dragging = false
			end
		end)
	end
end)

logoButton.InputChanged:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
		dragInput = input
	end
end)

UserInputService.InputChanged:Connect(function(input)
	if input == dragInput and dragging then
		local delta = input.Position - dragStart
		logoButton.Position = UDim2.new(
			startPos.X.Scale,
			startPos.X.Offset + delta.X,
			startPos.Y.Scale,
			startPos.Y.Offset + delta.Y
		)
	end
end)
